//+------------------------------------------------------------------+
//| Symbol Configuration Structure                                    |
//+------------------------------------------------------------------+
struct SymbolConfig
{
    string   symbol;
    double   multiplier;
    double   maxSpread;
    bool     isActive;
    
    // Trading state
    double   pivotPoint;
    double   vwap;
    double   keyLevel;
    double   pointOfInterest;
    bool     hasPosition;
    datetime lastTradeTime;
    
    // Fibonacci data cached from library
    bool     hasFibonacci;
    double   fib382;
    double   fib500;
    double   fib618;
};

//+------------------------------------------------------------------+
//| Global Variables                                                  |
//+------------------------------------------------------------------+
SymbolConfig    g_symbols[];
CPositionInfo   g_positionInfo;
int             g_totalPositions = 0;
bool            g_isFullyInitialized = false;
bool            g_initializationFailed = false;

//+------------------------------------------------------------------+
//| Validate Index Before Array Access                               |
//+------------------------------------------------------------------+
bool IsValidIndex(int index, string functionName = "")
{
    int arraySize = ArraySize(g_symbols);
    
    if(index < 0 || index >= arraySize || index >= g_symbolCount)
    {
        if(EnableDebugLogs)
            PrintFormat("ERROR [%s]: Invalid index %d (size: %d, count: %d)", 
                       functionName, index, arraySize, g_symbolCount);
        return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| OnInit - Initialization                                          |
//+------------------------------------------------------------------+
int OnInit()
{
    g_symbolCount = 0;
    g_isFullyInitialized = false;
    g_initializationFailed = false;
    g_totalPositions = 0;
    
    ArrayResize(g_symbols, 0);
    
    Print("========================================");
    Print("L'ACTION DE PRIX EA v3.04 - Starting...");
    Print("========================================");
    
    if(!InitializePriceAction())
    {
        Print("❌ Failed to initialize PriceAction library");
        g_initializationFailed = true;
        return INIT_FAILED;
    }
    Print("✓ Library initialized");
    
    if(!ConfigureSymbols())
    {
        Print("❌ Failed to configure symbols");
        g_symbolCount = 0;
        g_initializationFailed = true;
        return INIT_FAILED;
    }
    PrintFormat("✓ Configured %d symbols", g_symbolCount);
    
    // Load historical data
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(!IsValidIndex(i, "OnInit")) continue;
        
        string symbol = g_symbols[i].symbol;
        SymbolSelect(symbol, true);
        
        int bars = Bars(symbol, ATRTimeframe);
        if(bars < ATRPeriod + 10)
        {
            iTime(symbol, ATRTimeframe, ATRPeriod + 20);
            Sleep(200);
        }
    }
    Sleep(500);
    
    // Initialize in library
    int activeCount = 0;
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(!IsValidIndex(i, "OnInit")) continue;
        
        if(InitializeSymbol(g_symbols[i].symbol, g_symbols[i].multiplier, g_symbols[i].maxSpread))
        {
            g_symbols[i].isActive = true;
            activeCount++;
        }
    }
    
    if(activeCount == 0)
    {
        Print("❌ No symbols initialized");
        g_initializationFailed = true;
        return INIT_FAILED;
    }
    
    PrintFormat("✓ %d symbols active in library", activeCount);
    Sleep(500);
    
    InitializeATRHandles();
    
    g_trade.SetExpertMagicNumber(123456);
    g_trade.SetDeviationInPoints(10);
    g_trade.SetTypeFilling(ORDER_FILLING_IOC);
    
    EventSetTimer(30);
    
    g_isFullyInitialized = true;
    
    Print("========================================");
    Print("✅ EA READY TO TRADE");
    Print("========================================");
    PrintFormat("  Symbols: %d active", activeCount);
    PrintFormat("  Risk: %.2f%% | R:R: %.1f:1", RiskPercentage, RiskRewardRatio);
    PrintFormat("  Rejection Required: %s", RequireRejectionPattern ? "YES" : "NO");
    Print("========================================");
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| OnDeinit                                                          |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    g_isFullyInitialized = false;
    
    ReleaseATRHandles();
    DeinitializePriceAction();
    EventKillTimer();
    
    // Clean up chart objects
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(IsValidIndex(i, "OnDeinit"))
        {
            string symbol = g_symbols[i].symbol;
            ObjectDelete(0, "KeyLevel_" + symbol);
            ObjectDelete(0, "PointOfInterest_" + symbol);
            ObjectDelete(0, "Fib_" + symbol + "_Main");
        }
    }
    
    g_symbolCount = 0;
    ArrayResize(g_symbols, 0);
    
    PrintFormat("EA Deinitialized: %s", GetDeinitReasonText(reason));
}

//+------------------------------------------------------------------+
//| OnTimer - Update Key Levels                                      |
//+------------------------------------------------------------------+
void OnTimer()
{
    if(g_initializationFailed || !g_isFullyInitialized) return;
    if(g_symbolCount <= 0) return;
    
    CheckNewsImpact();
    
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(!IsValidIndex(i, "OnTimer")) continue;
        if(!g_symbols[i].isActive) continue;
        
        // Update key levels and POI
        CalculateKeyLevels(i);
        UpdateFibonacciFromLibrary(i);
        CalculatePointOfInterest(i);
        
        // Draw on chart
        DrawKeyLevel(i);
        DrawPointOfInterest(i);
    }
}

//+------------------------------------------------------------------+
//| OnTick - Main Trading Logic                                      |
//+------------------------------------------------------------------+
void OnTick()
{
    if(g_initializationFailed || !g_isFullyInitialized) return;
    if(g_symbolCount <= 0) return;
    
    g_totalPositions = GetActivePositionsCount();
    if(g_totalPositions >= MaxTotalPositions) return;
    
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(!IsValidIndex(i, "OnTick")) continue;
        if(!g_symbols[i].isActive) continue;
        
        string symbol = g_symbols[i].symbol;
        if(symbol == "" || symbol == NULL) continue;
        
        // Check existing positions
        int posCount = GetActivePositionsCount(symbol);
        g_symbols[i].hasPosition = (posCount >= MaxPositionsPerSymbol);
        
        if(g_symbols[i].hasPosition)
        {
            ModifyTrade(symbol);
            continue;
        }
        
        // Update levels
        CalculateKeyLevels(i);
        UpdateFibonacciFromLibrary(i);
        CalculatePointOfInterest(i);
        
        // Process in library (updates Fibonacci)
        ProcessSymbol(symbol, g_symbols[i].pointOfInterest);
        
        // Check trading conditions
        if(!IsTradingAllowed(symbol)) continue;
        if(!CheckTradeCooldown(i)) continue;
        
        // Get signal
        int signal = GetTradingSignalFromLibrary(i);
        if(signal == 0) continue;
        
        // Validate with rejection pattern if required
        if(RequireRejectionPattern)
        {
            if(!ValidateSignalWithKeyLevel(i, signal))
                continue;
        }
        
        // Execute trade
        if(ExecuteTrade(symbol, signal))
        {
            g_symbols[i].lastTradeTime = TimeCurrent();
            g_totalPositions++;
            
            PrintFormat("🎯 %s: Opened %s position at POI %.5f", 
                       symbol, (signal == 1 ? "BUY" : "SELL"), 
                       g_symbols[i].pointOfInterest);
        }
    }
}

//+------------------------------------------------------------------+
//| OnTrade - Position Updates                                       |
//+------------------------------------------------------------------+
void OnTrade()
{
    if(g_initializationFailed || !g_isFullyInitialized) return;
    
    g_totalPositions = GetActivePositionsCount();
    
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(!IsValidIndex(i, "OnTrade")) continue;
        if(!g_symbols[i].isActive) continue;
        
        string symbol = g_symbols[i].symbol;
        int posCount = GetActivePositionsCount(symbol);
        
        bool hadPosition = g_symbols[i].hasPosition;
        g_symbols[i].hasPosition = (posCount > 0);
        
        if(posCount == 0 && hadPosition)
        {
            PrintFormat("📊 %s: Position closed", symbol);
        }
    }
}

//+------------------------------------------------------------------+
//| Configure Symbols                                                |
//+------------------------------------------------------------------+
bool ConfigureSymbols()
{
    string symbols[] = {"EURUSD", "GBPUSD", "USDJPY", "AUDUSD", "USDCAD", "NZDUSD"};
    double multipliers[] = {10000, 10000, 100, 10000, 10000, 10000};
    double maxSpreads[] = {0.00030, 0.00035, 0.040, 0.00030, 0.00030, 0.00035};
    
    int count = ArraySize(symbols);
    if(count == 0) return false;
    
    if(ArrayResize(g_symbols, count) != count) return false;
    
    g_symbolCount = count;
    
    for(int i = 0; i < g_symbolCount; i++)
    {
        g_symbols[i].symbol = symbols[i];
        g_symbols[i].multiplier = multipliers[i];
        g_symbols[i].maxSpread = maxSpreads[i];
        g_symbols[i].isActive = false;
        g_symbols[i].pivotPoint = 0.0;
        g_symbols[i].vwap = 0.0;
        g_symbols[i].keyLevel = 0.0;
        g_symbols[i].pointOfInterest = 0.0;
        g_symbols[i].hasPosition = false;
        g_symbols[i].lastTradeTime = 0;
        g_symbols[i].hasFibonacci = false;
        g_symbols[i].fib382 = 0.0;
        g_symbols[i].fib500 = 0.0;
        g_symbols[i].fib618 = 0.0;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Calculate Key Levels (Pivot/VWAP)                                |
//+------------------------------------------------------------------+
void CalculateKeyLevels(int index)
{
    if(!IsValidIndex(index, "CalculateKeyLevels")) return;
    
    string symbol = g_symbols[index].symbol;
    
    double high = iHigh(symbol, KeyLevelTimeframe, 1);
    double low = iLow(symbol, KeyLevelTimeframe, 1);
    double close = iClose(symbol, KeyLevelTimeframe, 1);
    double open = iOpen(symbol, KeyLevelTimeframe, 1);
    
    if(high <= 0 || low <= 0 || close <= 0 || open <= 0) return;
    
    g_symbols[index].pivotPoint = (high + low + close) / 3.0;
    g_symbols[index].vwap = (high + low + close + open) / 4.0;
    
    if(UsePivotPoint && UseVWAP)
        g_symbols[index].keyLevel = (g_symbols[index].pivotPoint + g_symbols[index].vwap) / 2.0;
    else if(UsePivotPoint)
        g_symbols[index].keyLevel = g_symbols[index].pivotPoint;
    else if(UseVWAP)
        g_symbols[index].keyLevel = g_symbols[index].vwap;
    else
        g_symbols[index].keyLevel = g_symbols[index].pivotPoint;
}

//+------------------------------------------------------------------+
//| Update Fibonacci Data from Library                               |
//+------------------------------------------------------------------+
void UpdateFibonacciFromLibrary(int index)
{
    if(!IsValidIndex(index, "UpdateFibonacci")) return;
    
    string symbol = g_symbols[index].symbol;
    int symIndex = GetSymbolIndex(symbol);
    
    if(symIndex < 0 || symIndex >= MAX_SYMBOLS) return;
    
    // Get Fibonacci from library's g_symbolData
    FibonacciData fibData = g_symbolData[symIndex].fibData;
    
    if(fibData.isValid)
    {
        g_symbols[index].hasFibonacci = true;
        g_symbols[index].fib382 = fibData.fib382;
        g_symbols[index].fib500 = fibData.fib500;
        g_symbols[index].fib618 = fibData.fib618;
    }
    else
    {
        g_symbols[index].hasFibonacci = false;
    }
}

//+------------------------------------------------------------------+
//| Calculate Point of Interest                                      |
//+------------------------------------------------------------------+
void CalculatePointOfInterest(int index)
{
    if(!IsValidIndex(index, "CalculatePOI")) return;
    
    string symbol = g_symbols[index].symbol;
    double keyLevel = g_symbols[index].keyLevel;
    
    if(keyLevel <= 0)
    {
        g_symbols[index].pointOfInterest = 0.0;
        return;
    }
    
    // If we have Fibonacci data, find nearest level
    if(g_symbols[index].hasFibonacci)
    {
        double fibLevels[] = {g_symbols[index].fib382, g_symbols[index].fib500, g_symbols[index].fib618};
        double nearestFib = 0.0;
        double minDistance = DBL_MAX;
        
        for(int i = 0; i < 3; i++)
        {
            if(fibLevels[i] <= 0) continue;
            
            double distance = MathAbs(fibLevels[i] - keyLevel);
            if(distance < minDistance)
            {
                minDistance = distance;
                nearestFib = fibLevels[i];
            }
        }
        
        if(nearestFib > 0)
        {
            g_symbols[index].pointOfInterest = (keyLevel + nearestFib) / 2.0;
            
            if(EnableDebugLogs)
                PrintFormat("🎯 %s: POI = %.5f (KeyLevel: %.5f + Fib: %.5f)", 
                           symbol, g_symbols[index].pointOfInterest, keyLevel, nearestFib);
            return;
        }
    }
    
    // No Fibonacci - use key level as POI
    g_symbols[index].pointOfInterest = keyLevel;
    
    if(EnableDebugLogs)
        PrintFormat("🎯 %s: POI = %.5f (KeyLevel only)", symbol, keyLevel);
}

//+------------------------------------------------------------------+
//| Draw Key Level on Chart                                          |
//+------------------------------------------------------------------+
void DrawKeyLevel(int index)
{
    if(!IsValidIndex(index, "DrawKeyLevel")) return;
    
    string symbol = g_symbols[index].symbol;
    double keyLevel = g_symbols[index].keyLevel;
    
    if(keyLevel <= 0) return;
    
    // Only draw on current chart or all charts if enabled
    if(!DrawOnAllCharts && symbol != _Symbol) return;
    
    long chartID = DrawOnAllCharts ? ChartFirst() : 0;
    
    do
    {
        if(ChartSymbol(chartID) == symbol)
        {
            string lineName = "KeyLevel_" + symbol;
            ObjectDelete(chartID, lineName);
            
            if(ObjectCreate(chartID, lineName, OBJ_HLINE, 0, 0, keyLevel))
            {
                ObjectSetInteger(chartID, lineName, OBJPROP_COLOR, clrDodgerBlue);
                ObjectSetInteger(chartID, lineName, OBJPROP_STYLE, STYLE_DASHDOTDOT);
                ObjectSetInteger(chartID, lineName, OBJPROP_WIDTH, 2);
                ObjectSetString(chartID, lineName, OBJPROP_TEXT, "Key Level");
            }
        }
        
        if(!DrawOnAllCharts) break;
        chartID = ChartNext(chartID);
    }
    while(chartID >= 0);
}

//+------------------------------------------------------------------+
//| Draw Point of Interest on Chart                                  |
//+------------------------------------------------------------------+
void DrawPointOfInterest(int index)
{
    if(!IsValidIndex(index, "DrawPOI")) return;
    
    string symbol = g_symbols[index].symbol;
    double poi = g_symbols[index].pointOfInterest;
    
    if(poi <= 0) return;
    
    if(!DrawOnAllCharts && symbol != _Symbol) return;
    
    long chartID = DrawOnAllCharts ? ChartFirst() : 0;
    
    do
    {
        if(ChartSymbol(chartID) == symbol)
        {
            string lineName = "PointOfInterest_" + symbol;
            ObjectDelete(chartID, lineName);
            
            if(ObjectCreate(chartID, lineName, OBJ_HLINE, 0, 0, poi))
            {
                ObjectSetInteger(chartID, lineName, OBJPROP_COLOR, clrOrange);
                ObjectSetInteger(chartID, lineName, OBJPROP_STYLE, STYLE_SOLID);
                ObjectSetInteger(chartID, lineName, OBJPROP_WIDTH, 3);
                ObjectSetString(chartID, lineName, OBJPROP_TEXT, "Point of Interest");
            }
        }
        
        if(!DrawOnAllCharts) break;
        chartID = ChartNext(chartID);
    }
    while(chartID >= 0);
}

//+------------------------------------------------------------------+
//| Get Trading Signal from Library                                  |
//+------------------------------------------------------------------+
int GetTradingSignalFromLibrary(int index)
{
    if(!IsValidIndex(index, "GetSignal")) return 0;
    
    string symbol = g_symbols[index].symbol;
    double poi = g_symbols[index].pointOfInterest;
    
    if(poi <= 0) return 0;
    
    // Check if price is near POI
    if(!IsPriceNearLevel(symbol, poi, KeyLevelTolerance))
    {
        if(EnableDebugLogs)
            LogDebug(StringFormat("%s: Price not near POI %.5f", symbol, poi));
        return 0;
    }
    
    // Get Fibonacci direction from library
    int symIndex = GetSymbolIndex(symbol);
    if(symIndex < 0) return 0;
    
    bool isBullishFib = g_symbolData[symIndex].isBullishFib;
    bool isBearishFib = g_symbolData[symIndex].isBearishFib;
    
    if(isBullishFib) return 1;   // Buy signal
    if(isBearishFib) return -1;  // Sell signal
    
    return 0;
}

//+------------------------------------------------------------------+
//| Validate Signal with Rejection Pattern                           |
//+------------------------------------------------------------------+
bool ValidateSignalWithKeyLevel(int index, int signal)
{
    if(!IsValidIndex(index, "ValidateSignal")) return false;
    
    string symbol = g_symbols[index].symbol;
    double poi = g_symbols[index].pointOfInterest;
    
    if(poi <= 0) return false;
    
    // Check for rejection pattern at POI
    RejectionSignal rejection = PricePosition(symbol, poi, PatternTimeframe);
    
    if(signal == 1)  // Buy signal
    {
        if(!rejection.isBullishRejection)
        {
            if(EnableDebugLogs)
                LogDebug(StringFormat("%s: No bullish rejection at POI", symbol));
            return false;
        }
        
        PrintFormat("✅ %s: Bullish rejection confirmed - %s", symbol, rejection.rejectionType);
        return true;
    }
    else if(signal == -1)  // Sell signal
    {
        if(!rejection.isBearishRejection)
        {
            if(EnableDebugLogs)
                LogDebug(StringFormat("%s: No bearish rejection at POI", symbol));
            return false;
        }
        
        PrintFormat("✅ %s: Bearish rejection confirmed - %s", symbol, rejection.rejectionType);
        return true;
    }
    
    return false;
}

//+------------------------------------------------------------------+
//| Check Trade Cooldown                                             |
//+------------------------------------------------------------------+
bool CheckTradeCooldown(int index)
{
    if(!IsValidIndex(index, "CheckCooldown")) return false;
    
    if(g_symbols[index].lastTradeTime == 0) return true;
    
    datetime timeSinceLast = TimeCurrent() - g_symbols[index].lastTradeTime;
    int cooldownSeconds = MinutesBetweenTrades * 60;
    
    if(timeSinceLast < cooldownSeconds)
    {
        if(EnableDebugLogs)
            LogDebug(StringFormat("%s: Cooldown - %d sec remaining", 
                                 g_symbols[index].symbol, 
                                 cooldownSeconds - (int)timeSinceLast));
        return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Get Deinitialization Reason                                      |
//+------------------------------------------------------------------+
string GetDeinitReasonText(int reason)
{
    switch(reason)
    {
        case REASON_REMOVE:      return "EA removed";
        case REASON_RECOMPILE:   return "Recompiled";
        case REASON_CHARTCHANGE: return "Chart changed";
        case REASON_CHARTCLOSE:  return "Chart closed";
        case REASON_PARAMETERS:  return "Parameters changed";
        case REASON_ACCOUNT:     return "Account changed";
        case REASON_TEMPLATE:    return "Template applied";
        case REASON_INITFAILED:  return "Init failed";
        case REASON_CLOSE:       return "Terminal closed";
        default:                 return "Unknown";
    }
}
