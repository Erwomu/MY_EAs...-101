//+------------------------------------------------------------------+
//| Expert Initialization Function                                    |
//+------------------------------------------------------------------+
int OnInit()
{
    // Initialize PriceAction library first
    if(!InitializePriceAction())
    {
        Print("Error: Failed to initialize PriceAction library");
        return INIT_FAILED;
    }
    
    // Configure symbols locally
    ConfigureSymbols();
    
    // Initialize each symbol in the PriceAction library
    for(int i = 0; i < ArraySize(g_symbols); i++)
    {
        if(!InitializeSymbol(g_symbols[i].symbol, g_symbols[i].multiplier, g_symbols[i].maxSpread))
        {
            PrintFormat("Warning: Failed to initialize %s in PriceAction library", g_symbols[i].symbol);
            g_symbols[i].isActive = false;
        }
        else
        {
            g_symbols[i].isActive = true;
            PrintFormat("Symbol %s initialized successfully", g_symbols[i].symbol);
        }
    }
    
    // NOW resize pattern arrays based on YOUR symbol count
    int localSymbolCount = ArraySize(g_symbols);
    ArrayResize(g_bullishEngulfing, localSymbolCount);
    ArrayResize(g_bearishEngulfing, localSymbolCount);
    ArrayResize(g_pinBarSignal, localSymbolCount);
    
    // Set up trading object
    g_trade.SetExpertMagicNumber(123456);
    g_trade.SetDeviationInPoints(10);
    g_trade.SetTypeFilling(ORDER_FILLING_IOC);
    g_trade.SetTypeFillingBySymbol(_Symbol);
    
    // Set timer for periodic updates
    EventSetTimer(30);
    
    Print("=========================================");
    Print("L'ACTION DE PRIX EA v2.0 Initialized");
    Print("Active Symbols: ", CountActiveSymbols());
    Print("Risk Per Trade: ", RiskPercent, "%");
    Print("=========================================");
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| Expert Deinitialization Function                                  |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    // Clean up PriceAction library
    DeinitializePriceAction();
    
    // Kill timer
    EventKillTimer();
    
    // Log deinitialization reason
    string reasonText = GetDeinitReasonText(reason);
    PrintFormat("EA Deinitialized: %s", reasonText);
    
    // Clean up chart objects
    ObjectsDeleteAll(0, 0, -1);
    ChartRedraw();
}

//+------------------------------------------------------------------+
//| Timer Event Function                                              |
//+------------------------------------------------------------------+
void OnTimer()
{
    // Check news impact
    if(UseNewsFilter)
    {
        CheckNewsImpact();
    }
    
    // Update each symbol's analysis
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(!g_symbols[i].isActive)
            continue;
            
        string symbol = g_symbols[i].symbol;
        
        // Process symbol through PriceAction library
        ProcessSymbol(symbol);
        
        // Calculate key levels
        CalculateKeyLevels(i);
        
        // Draw Fibonacci retracement
        DrawFibonacci(symbol);
        
        // Find matching Fibonacci level
        g_symbols[i].fibLevel = FindMatchingFibLevel(i);
        
        // Draw support/resistance zones
        if(g_symbols[i].fibLevel > 0 && g_symbols[i].keyLevel > 0)
        {
            DrawZone(g_symbols[i].fibLevel, g_symbols[i].keyLevel, symbol);
        }
        
        // Calculate lot size
        g_symbols[i].lotSize = CalculateLotSize(symbol, GetSymbolIndex(symbol));
        
        // Log symbol status
        LogDebug(GetSymbolStatus(symbol));
    }
}

//+------------------------------------------------------------------+
//| Tick Event Function                                               |
//+------------------------------------------------------------------+
void OnTick()
{
    // Update total positions count
    g_totalPositions = GetActivePositionsCount();
    
    // Check if trading is allowed globally
    if(!g_tradingEnabled || g_totalPositions >= MaxTotalPositions)
        return;
        
    // Process each symbol
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(!g_symbols[i].isActive)
            continue;
            
        string symbol = g_symbols[i].symbol;
        
        // Skip if symbol already has position
        if(GetActivePositionsCount(symbol) >= MaxPositionsPerSymbol)
        {
            g_symbols[i].hasPosition = true;
            continue;
        }
        else
        {
            g_symbols[i].hasPosition = false;
        }
        
        // Check trading conditions
        if(!CheckTradingConditions(i))
            continue;
            
        // Detect patterns
        DetectPatterns(i);
        
        // Check price position
        double keyLevel = (g_symbols[i].fibLevel + g_symbols[i].keyLevel) / 2;
        if(keyLevel > 0)
        {
            PricePosition(keyLevel, symbol);
        }
        
        // Generate trading signals
        GenerateSignals(i);
        
        // Execute trades if signals are valid
        if(g_symbols[i].bullishSignal && ValidateBuySignal(i))
        {
            ExecuteBuyTrade(i);
        }
        else if(g_symbols[i].sellSignal && ValidateSellSignal(i))
        {
            ExecuteSellTrade(i);
        }
    }
    
    // Manage existing positions
    ManageOpenPositions();
}

//+------------------------------------------------------------------+
//| Calculate Key Levels (Pivot/VWAP)                                |
//+------------------------------------------------------------------+
void CalculateKeyLevels(int index)
{
    string symbol = g_symbols[index].symbol;
    
    double high = iHigh(symbol, PivotTimeframe, 1);
    double low = iLow(symbol, PivotTimeframe, 1);
    double close = iClose(symbol, PivotTimeframe, 1);
    double open = iOpen(symbol, PivotTimeframe, 1);
    
    if(high <= 0 || low <= 0 || close <= 0)
    {
        LogDebug("Failed to retrieve price data for " + symbol);
        return;
    }
    
    // Calculate Pivot Point
    g_symbols[index].pivotPoint = (high + low + close) / 3;
    
    // Calculate VWAP
    g_symbols[index].vwap = (high + low + close + open) / 4;
    
    // Set key level based on selection
    if(PointOfInterest == PIVOT_POINT)
        g_symbols[index].keyLevel = g_symbols[index].pivotPoint;
    else
        g_symbols[index].keyLevel = g_symbols[index].vwap;
        
    // Draw horizontal line for key level
    string lineName = "KeyLevel_" + symbol;
    ObjectDelete(0, lineName);
    
    if(g_symbols[index].keyLevel > 0)
    {
        ObjectCreate(0, lineName, OBJ_HLINE, 0, 0, g_symbols[index].keyLevel);
        ObjectSetInteger(0, lineName, OBJPROP_COLOR, clrGold);
        ObjectSetInteger(0, lineName, OBJPROP_STYLE, STYLE_DASH);
        ObjectSetInteger(0, lineName, OBJPROP_WIDTH, 2);
    }
}

//+------------------------------------------------------------------+
//| Find Matching Fibonacci Level                                    |
//+------------------------------------------------------------------+
double FindMatchingFibLevel(int index)
{
    string symbol = g_symbols[index].symbol;
    int symIndex = GetSymbolIndex(symbol);
    
    if(symIndex < 0)
        return 0.0;
        
    // Get Fibonacci data from PriceAction library
    double highPrice = g_symbolData[symIndex].highestPrice;
    double lowPrice = g_symbolData[symIndex].lowestPrice;
    bool isBullish = g_symbolData[symIndex].isBullishFib;
    bool isBearish = g_symbolData[symIndex].isBearishFib;
    
    if(highPrice <= 0 || lowPrice <= 0 || (!isBullish && !isBearish))
        return 0.0;
        
    double fibLevels[4];
    double keyLevel = g_symbols[index].keyLevel;
    
    // Calculate Fibonacci levels
    if(isBullish)
    {
        fibLevels[0] = lowPrice + (highPrice - lowPrice) * 0.236;
        fibLevels[1] = lowPrice + (highPrice - lowPrice) * 0.382;
        fibLevels[2] = lowPrice + (highPrice - lowPrice) * 0.500;
        fibLevels[3] = lowPrice + (highPrice - lowPrice) * 0.618;
    }
    else
    {
        fibLevels[0] = highPrice - (highPrice - lowPrice) * 0.236;
        fibLevels[1] = highPrice - (highPrice - lowPrice) * 0.382;
        fibLevels[2] = highPrice - (highPrice - lowPrice) * 0.500;
        fibLevels[3] = highPrice - (highPrice - lowPrice) * 0.618;
    }
    
    // Find closest Fibonacci level to key level
    double minDiff = DBL_MAX;
    double closestLevel = 0.0;
    double tolerance = GetTolerance(symbol);
    
    for(int i = 0; i < 4; i++)
    {
        double diff = MathAbs(fibLevels[i] - keyLevel);
        if(diff < minDiff && diff <= tolerance)
        {
            minDiff = diff;
            closestLevel = fibLevels[i];
        }
    }
    
    return closestLevel;
}

//+------------------------------------------------------------------+
//| Get Tolerance for Symbol                                         |
//+------------------------------------------------------------------+
double GetTolerance(string symbol)
{
    if(symbol == "BTCUSD") return 200.0;
    if(symbol == "GOLD") return 2.5;
    if(symbol == "USDJPY") return 0.20;
    return 0.0015; // Default for forex pairs
}

//+------------------------------------------------------------------+
//| Detect Candlestick Patterns                                      |
//+------------------------------------------------------------------+
bool DetectBullishEngulfing(string symbol, ENUM_TIMEFRAMES timeframe)
{
    double open1 = iOpen(symbol, timeframe, 1);
    double close1 = iClose(symbol, timeframe, 1);
    double high1 = iHigh(symbol, timeframe, 1);
    double low1 = iLow(symbol, timeframe, 1);
    
    double open2 = iOpen(symbol, timeframe, 2);
    double close2 = iClose(symbol, timeframe, 2);
    double high2 = iHigh(symbol, timeframe, 2);
    double low2 = iLow(symbol, timeframe, 2);
    
    if(open1 <= 0 || close1 <= 0 || open2 <= 0 || close2 <= 0)
        return false;
    
    // Calculate body sizes
    double body1 = MathAbs(close1 - open1);
    double body2 = MathAbs(close2 - open2);
    
    if(body1 <= 0 || body2 <= 0)
        return false;
    
    // Previous candle must be bearish
    if(close2 >= open2)
        return false;
    
    // Current candle must be bullish
    if(close1 <= open1)
        return false;
    
    // REALISTIC CONDITIONS:
    // 1. Current body is at least 80% of previous body
    if(body1 < body2 * 0.8)
        return false;
    
    // 2. Current candle closes above midpoint of previous candle
    double midpoint2 = (open2 + close2) / 2.0;
    if(close1 <= midpoint2)
        return false;
    
    // 3. Current candle opens below midpoint of previous candle
    if(open1 >= midpoint2)
        return false;
    
    // 4. Significant range (not tiny candles)
    double avgBody = (body1 + body2) / 2.0;
    double minBodySize = g_symbolInfo.Point() * 50;  // At least 5 pips
    
    if(avgBody < minBodySize)
        return false;
    
    LogDebug(StringFormat("%s: Bullish engulfing detected. Body1: %.5f, Body2: %.5f", 
                         symbol, body1, body2));
    
    return true;
}
//+------------------------------------------------------------------+
//| Detect Pin Bar Pattern                                           |
//+------------------------------------------------------------------+
int DetectPinBar(string symbol, ENUM_TIMEFRAMES timeframe, int shift)
{
    double open = iOpen(symbol, timeframe, shift);
    double close = iClose(symbol, timeframe, shift);
    double high = iHigh(symbol, timeframe, shift);
    double low = iLow(symbol, timeframe, shift);
    
    if(open <= 0 || close <= 0 || high <= 0 || low <= 0)
        return 0;
        
    double bodySize = MathAbs(close - open);
    double upperWick = high - MathMax(open, close);
    double lowerWick = MathMin(open, close) - low;
    
    if(bodySize <= 0)
        return 0;
        
    // Bullish pin bar (long lower wick)
    if(lowerWick > bodySize * 2.0 && lowerWick > upperWick && close > open)
        return 1;
        
    // Bearish pin bar (long upper wick)
    if(upperWick > bodySize * 2.0 && upperWick > lowerWick && close < open)
        return -1;
        
    return 0;
}

//+------------------------------------------------------------------+
//| Generate Trading Signals                                         |
//+------------------------------------------------------------------+
void GenerateSignals(int index)
{
    string symbol = g_symbols[index].symbol;
    int symIndex = GetSymbolIndex(symbol);
    
    g_symbols[index].bullishSignal = false;
    g_symbols[index].sellSignal = false;
    
    if(symIndex < 0)
        return;
        
    // Get signals from PriceAction library
    bool priceAbove = g_symbolData[symIndex].priceIsAbove;
    bool priceBelow = g_symbolData[symIndex].priceIsBelow;
    bool bullishFib = g_symbolData[symIndex].isBullishFib;
    bool bearishFib = g_symbolData[symIndex].isBearishFib;
    
    // Generate buy signal
    if(bullishFib && priceAbove)
    {
        if(!RequireEngulfing || g_bullishEngulfing[index])
        {
            if(!RequirePinBar || g_pinBarSignal[index] == 1)
            {
                g_symbols[index].bullishSignal = true;
                LogDebug(symbol + ": Buy signal generated");
            }
        }
    }
    
    // Generate sell signal
    if(bearishFib && priceBelow)
    {
        if(!RequireEngulfing || g_bearishEngulfing[index])
        {
            if(!RequirePinBar || g_pinBarSignal[index] == -1)
            {
                g_symbols[index].sellSignal = true;
                LogDebug(symbol + ": Sell signal generated");
            }
        }
    }
}

//+------------------------------------------------------------------+
//| Execute Buy Trade                                                |
//+------------------------------------------------------------------+
bool ExecuteBuyTrade(int index)
{
    string symbol = g_symbols[index].symbol;
    
    if(!g_symbolInfo.Name(symbol))
        return false;
        
    g_symbolInfo.RefreshRates();
    
    double price = g_symbolInfo.Ask();
    double stopLoss = NormalizeDouble(price - g_symbols[index].stopLoss, g_symbolInfo.Digits());
    double takeProfit = NormalizeDouble(price + (g_symbols[index].stopLoss * RiskRewardRatio), g_symbolInfo.Digits());
    double lotSize = g_symbols[index].lotSize;
    
    // Validate levels
    if(stopLoss >= price || takeProfit <= price)
    {
        Print(symbol, ": Invalid SL/TP levels for buy trade");
        return false;
    }
    
    // Execute trade
    if(g_trade.Buy(lotSize, symbol, price, stopLoss, takeProfit, "PA_BUY_" + symbol))
    {
        if(g_trade.ResultRetcode() == TRADE_RETCODE_DONE)
        {
            g_symbols[index].lastTradeTime = TimeCurrent();
            g_symbols[index].currentSL = stopLoss;
            g_symbols[index].currentTP = takeProfit;
            
            PrintFormat("✅ BUY %s: Price=%.5f SL=%.5f TP=%.5f Lot=%.2f",
                       symbol, price, stopLoss, takeProfit, lotSize);
            return true;
        }
    }
    
    PrintFormat("❌ Failed to open BUY %s: Error %d", symbol, g_trade.ResultRetcode());
    return false;
}

//+------------------------------------------------------------------+
//| Execute Sell Trade                                               |
//+------------------------------------------------------------------+
bool ExecuteSellTrade(int index)
{
    string symbol = g_symbols[index].symbol;
    
    if(!g_symbolInfo.Name(symbol))
        return false;
        
    g_symbolInfo.RefreshRates();
    
    double price = g_symbolInfo.Bid();
    double stopLoss = NormalizeDouble(price + g_symbols[index].stopLoss, g_symbolInfo.Digits());
    double takeProfit = NormalizeDouble(price - (g_symbols[index].stopLoss * RiskRewardRatio), g_symbolInfo.Digits());
    double lotSize = g_symbols[index].lotSize;
    
    // Validate levels
    if(stopLoss <= price || takeProfit >= price)
    {
        Print(symbol, ": Invalid SL/TP levels for sell trade");
        return false;
    }
    
    // Execute trade
    if(g_trade.Sell(lotSize, symbol, price, stopLoss, takeProfit, "PA_SELL_" + symbol))
    {
        if(g_trade.ResultRetcode() == TRADE_RETCODE_DONE)
        {
            g_symbols[index].lastTradeTime = TimeCurrent();
            g_symbols[index].currentSL = stopLoss;
            g_symbols[index].currentTP = takeProfit;
            
            PrintFormat("✅ SELL %s: Price=%.5f SL=%.5f TP=%.5f Lot=%.2f",
                       symbol, price, stopLoss, takeProfit, lotSize);
            return true;
        }
    }
    
    PrintFormat("❌ Failed to open SELL %s: Error %d", symbol, g_trade.ResultRetcode());
    return false;
}

//+------------------------------------------------------------------+
//| Check Trading Conditions                                         |
//+------------------------------------------------------------------+
bool CheckTradingConditions(int index)
{
    string symbol = g_symbols[index].symbol;
    
    // Check if news filter blocks trading
    if(g_newsImpactActive)
        return false;
        
    // Check trading hours
    if(UseTradingHours && !IsWithinTradingHours())
        return false;
        
    // Check spread
    if(UseSpreadFilter && !IsSpreadAcceptable(symbol, g_symbols[index].maxSpread))
        return false;
        
    // Check if symbol is paused due to losses
    int symIndex = GetSymbolIndex(symbol);
    if(symIndex >= 0 && g_symbolData[symIndex].tradingPaused)
        return false;
        
    return true;
}

//+------------------------------------------------------------------+
//| Manage Open Positions                                            |
//+------------------------------------------------------------------+
void ManageOpenPositions()
{
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(!g_symbols[i].isActive || !g_symbols[i].hasPosition)
            continue;
            
        string symbol = g_symbols[i].symbol;
        
        // Use PriceAction library's position modification
        ModifyTrade(symbol);
        
        // Check for consecutive losses
        ManageTradingPause(symbol);
    }
}

//+------------------------------------------------------------------+
//| Count Active Symbols                                             |
//+------------------------------------------------------------------+
int CountActiveSymbols()
{
    int count = 0;
    for(int i = 0; i < g_symbolCount; i++)
    {
        if(g_symbols[i].isActive)
            count++;
    }
    return count;
}

    }
}

{

    g_totalPositions = GetActivePositionsCount();
    
    // Check each symbol for closed positions
    for(int i = 0; i < g_symbolCount; i++)
    {
        string symbol = g_symbols[i].symbol;

        }
    }
}
