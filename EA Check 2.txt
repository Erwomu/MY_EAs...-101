input int                MaxTotalPositions  = 3;              // Max total positions across all symbols

sinput string            SEP3                = "=== Pattern Requirements ===";
input bool               RequireRejectionPattern = false;     // Require price rejection pattern (set to FALSE for more trades)
input double             KeyLevelTolerance  = 15.0;          // Key level proximity (pips) - INCREASED from 5 to 15

sinput string            SEP4                = "=== Trading Cooldown ===";
input int                MinutesBetweenTrades = 5;            // Minimum minutes between trades per symbol

//+------------------------------------------------------------------+
//| Symbol Configuration Structure                                    |
//+------------------------------------------------------------------+
struct SymbolConfig
{
    string   symbol;
    double   multiplier;
    double   maxSpread;
    bool     isActive;
    
    // Trading state
    double   pivotPoint;
    double   vwap;
    double   keyLevel;
    double   pointOfInterest;
    bool     hasPosition;
    datetime lastTradeTime;
    
    // Cached library index
    int      libraryIndex;
};

//+------------------------------------------------------------------+
//| Global Variables                                                  |
//+------------------------------------------------------------------+
SymbolConfig    g_symbols[];
int             g_eaSymbolCount = 0;      // EA's own symbol count - RENAMED!
CPositionInfo   g_positionInfo;
int             g_totalPositions = 0;
bool            g_isFullyInitialized = false;
bool            g_initializationFailed = false;

//+------------------------------------------------------------------+
//| Validate Index                                                    |
//+------------------------------------------------------------------+
bool IsValidIndex(int index, string functionName = "")
{
    int arraySize = ArraySize(g_symbols);
    
    if(index < 0 || index >= arraySize || index >= g_eaSymbolCount)
    {
        PrintFormat("ERROR [%s]: Invalid index %d (size: %d, EA count: %d)", 
                   functionName, index, arraySize, g_eaSymbolCount);
        return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| OnInit                                                            |
//+------------------------------------------------------------------+
int OnInit()
{
    g_eaSymbolCount = 0;  // Use EA's own counter
    g_isFullyInitialized = false;
    g_initializationFailed = false;
    g_totalPositions = 0;
    
    ArrayResize(g_symbols, 0);
    
    Print("========================================");
    Print("L'ACTION DE PRIX EA v3.05");
    Print("========================================");
    
    // Initialize library FIRST (resets library's g_symbolCount)
    if(!InitializePriceAction())
    {
        Print("❌ Library init failed");
        g_initializationFailed = true;
        return INIT_FAILED;
    }
    Print("✓ Library initialized");
    
    // Configure EA's symbols
    if(!ConfigureSymbols())
    {
        Print("❌ Symbol config failed");
        g_eaSymbolCount = 0;
        g_initializationFailed = true;
        return INIT_FAILED;
    }
    PrintFormat("✓ EA: Configured %d symbols", g_eaSymbolCount);
    
    // Load historical data
    for(int i = 0; i < g_eaSymbolCount; i++)
    {
        if(!IsValidIndex(i, "OnInit")) continue;
        
        string symbol = g_symbols[i].symbol;
        SymbolSelect(symbol, true);
        
        int bars = Bars(symbol, ATRTimeframe);
        if(bars < ATRPeriod + 10)
        {
            iTime(symbol, ATRTimeframe, ATRPeriod + 20);
            Sleep(200);
        }
    }
    Sleep(500);
    
    // Initialize in library and map indices
    int activeCount = 0;
    for(int i = 0; i < g_eaSymbolCount; i++)
    {
        if(!IsValidIndex(i, "OnInit")) continue;
        
        // Initialize in library
        if(InitializeSymbol(g_symbols[i].symbol, g_symbols[i].multiplier, g_symbols[i].maxSpread))
        {
            // Get library index for this symbol
            g_symbols[i].libraryIndex = GetSymbolIndex(g_symbols[i].symbol);
            g_symbols[i].isActive = (g_symbols[i].libraryIndex >= 0);
            
            if(g_symbols[i].isActive)
            {
                activeCount++;
                PrintFormat("  ✓ %s -> Library index %d", g_symbols[i].symbol, g_symbols[i].libraryIndex);
            }
        }
        else
        {
            g_symbols[i].isActive = false;
            g_symbols[i].libraryIndex = -1;
        }
    }
    
    if(activeCount == 0)
    {
        Print("❌ No active symbols");
        g_initializationFailed = true;
        return INIT_FAILED;
    }
    
    PrintFormat("✓ %d symbols active in library", activeCount);
    Sleep(500);
    
    InitializeATRHandles();
    
    g_trade.SetExpertMagicNumber(123456);
    g_trade.SetDeviationInPoints(10);
    g_trade.SetTypeFilling(ORDER_FILLING_IOC);
    
    EventSetTimer(30);
    
    g_isFullyInitialized = true;
    
    Print("========================================");
    Print("✅ EA READY");
    Print("========================================");
    PrintFormat("  Symbols: %d/%d active", activeCount, g_eaSymbolCount);
    PrintFormat("  Risk: %.2f%% | R:R: %.1f:1", RiskPercentage, RiskRewardRatio);
    PrintFormat("  Rejection: %s", RequireRejectionPattern ? "Required" : "Optional");
    PrintFormat("  POI Tolerance: %.0f pips", KeyLevelTolerance);
    Print("========================================");
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| OnDeinit                                                          |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    g_isFullyInitialized = false;
    
    // Clean up objects
    for(int i = 0; i < g_eaSymbolCount; i++)
    {
        if(IsValidIndex(i, "OnDeinit"))
        {
            string symbol = g_symbols[i].symbol;
            ObjectDelete(0, "KeyLevel_" + symbol);
            ObjectDelete(0, "PointOfInterest_" + symbol);
            string prefix = "Fib_" + symbol + "_";
            ObjectDelete(0, prefix + "Main");
            ObjectDelete(0, prefix + "HighMarker");
            ObjectDelete(0, prefix + "LowMarker");
        }
    }
    
    ReleaseATRHandles();
    DeinitializePriceAction();
    EventKillTimer();
    
    g_eaSymbolCount = 0;
    ArrayResize(g_symbols, 0);
    
    PrintFormat("EA Deinitialized: %s", GetDeinitReasonText(reason));
}

//+------------------------------------------------------------------+
//| OnTimer                                                           |
//+------------------------------------------------------------------+
void OnTimer()
{
    if(g_initializationFailed || !g_isFullyInitialized) return;
    if(g_eaSymbolCount <= 0) return;
    
    CheckNewsImpact();
    
    for(int i = 0; i < g_eaSymbolCount; i++)
    {
        if(!IsValidIndex(i, "OnTimer")) continue;
        if(!g_symbols[i].isActive) continue;
        
        string symbol = g_symbols[i].symbol;
        
        // Calculate levels
        CalculateKeyLevels(i);
        
        // Process in library (calculates Fibonacci)
        ProcessSymbol(symbol, 0.0);
        
        // Update Fibonacci from library
        UpdateFibonacciFromLibrary(i);
        
        // Calculate POI
        CalculatePointOfInterest(i);
        
        // Draw everything
        DrawKeyLevel(i);
        DrawPointOfInterest(i);
        DrawFibonacci(i);
    }
}

//+------------------------------------------------------------------+
//| OnTick                                                            |
//+------------------------------------------------------------------+
void OnTick()
{
    if(g_initializationFailed || !g_isFullyInitialized) return;
    if(g_eaSymbolCount <= 0) return;
    
    g_totalPositions = GetActivePositionsCount();
    if(g_totalPositions >= MaxTotalPositions) return;
    
    for(int i = 0; i < g_eaSymbolCount; i++)
    {
        if(!IsValidIndex(i, "OnTick")) continue;
        if(!g_symbols[i].isActive) continue;
        
        string symbol = g_symbols[i].symbol;
        
        // Check existing positions
        int posCount = GetActivePositionsCount(symbol);
        g_symbols[i].hasPosition = (posCount >= MaxPositionsPerSymbol);
        
        if(g_symbols[i].hasPosition)
        {
            ModifyTrade(symbol);
            continue;
        }
        
        // Update everything
        CalculateKeyLevels(i);
        ProcessSymbol(symbol, 0.0);
        UpdateFibonacciFromLibrary(i);
        CalculatePointOfInterest(i);
        
        // Check trading conditions
        if(!IsTradingAllowed(symbol)) continue;
        if(!CheckTradeCooldown(i)) continue;
        
        // Get signal
        int signal = GetTradingSignal(i);
        if(signal == 0) continue;
        
        // Execute
        if(ExecuteTrade(symbol, signal))
        {
            g_symbols[i].lastTradeTime = TimeCurrent();
            g_totalPositions++;
            
            PrintFormat("🎯 %s: %s at %.5f (POI: %.5f)", 
                       symbol, (signal == 1 ? "BUY" : "SELL"), 
                       SymbolInfoDouble(symbol, SYMBOL_BID),
                       g_symbols[i].pointOfInterest);
        }
    }
}

//+------------------------------------------------------------------+
//| OnTrade                                                           |
//+------------------------------------------------------------------+
void OnTrade()
{
    if(g_initializationFailed || !g_isFullyInitialized) return;
    
    g_totalPositions = GetActivePositionsCount();
    
    for(int i = 0; i < g_eaSymbolCount; i++)
    {
        if(!IsValidIndex(i, "OnTrade")) continue;
        if(!g_symbols[i].isActive) continue;
        
        string symbol = g_symbols[i].symbol;
        int posCount = GetActivePositionsCount(symbol);
        
        bool hadPosition = g_symbols[i].hasPosition;
        g_symbols[i].hasPosition = (posCount > 0);
        
        if(posCount == 0 && hadPosition)
        {
            PrintFormat("📊 %s: Position closed", symbol);
        }
    }
}

//+------------------------------------------------------------------+
//| Configure Symbols                                                |
//+------------------------------------------------------------------+
bool ConfigureSymbols()
{
    string symbols[] = {"EURUSD", "GBPUSD", "USDJPY", "AUDUSD", "USDCAD", "NZDUSD"};
    double multipliers[] = {10000, 10000, 100, 10000, 10000, 10000};
    double maxSpreads[] = {0.00030, 0.00035, 0.040, 0.00030, 0.00030, 0.00035};
    
    int count = ArraySize(symbols);
    if(count == 0) return false;
    
    if(ArrayResize(g_symbols, count) != count) return false;
    
    g_eaSymbolCount = count;  // Set EA's counter
    
    for(int i = 0; i < g_eaSymbolCount; i++)
    {
        g_symbols[i].symbol = symbols[i];
        g_symbols[i].multiplier = multipliers[i];
        g_symbols[i].maxSpread = maxSpreads[i];
        g_symbols[i].isActive = false;
        g_symbols[i].pivotPoint = 0.0;
        g_symbols[i].vwap = 0.0;
        g_symbols[i].keyLevel = 0.0;
        g_symbols[i].pointOfInterest = 0.0;
        g_symbols[i].hasPosition = false;
        g_symbols[i].lastTradeTime = 0;
        g_symbols[i].libraryIndex = -1;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Calculate Key Levels                                             |
//+------------------------------------------------------------------+
void CalculateKeyLevels(int index)
{
    if(!IsValidIndex(index, "CalculateKeyLevels")) return;
    
    string symbol = g_symbols[index].symbol;
    
    double high = iHigh(symbol, KeyLevelTimeframe, 1);
    double low = iLow(symbol, KeyLevelTimeframe, 1);
    double close = iClose(symbol, KeyLevelTimeframe, 1);
    double open = iOpen(symbol, KeyLevelTimeframe, 1);
    
    if(high <= 0 || low <= 0 || close <= 0 || open <= 0) return;
    
    g_symbols[index].pivotPoint = (high + low + close) / 3.0;
    g_symbols[index].vwap = (high + low + close + open) / 4.0;
    
    if(UsePivotPoint && UseVWAP)
        g_symbols[index].keyLevel = (g_symbols[index].pivotPoint + g_symbols[index].vwap) / 2.0;
    else if(UsePivotPoint)
        g_symbols[index].keyLevel = g_symbols[index].pivotPoint;
    else if(UseVWAP)
        g_symbols[index].keyLevel = g_symbols[index].vwap;
    else
        g_symbols[index].keyLevel = g_symbols[index].pivotPoint;
}

//+------------------------------------------------------------------+
//| Update Fibonacci from Library                                    |
//+------------------------------------------------------------------+
void UpdateFibonacciFromLibrary(int index)
{
    if(!IsValidIndex(index, "UpdateFib")) return;
    
    int libIndex = g_symbols[index].libraryIndex;
    if(libIndex < 0 || libIndex >= MAX_SYMBOLS) return;
    
    // Access library's data
    FibonacciData fibData = g_symbolData[libIndex].fibData;
    
    // Store validity for POI calculation
    g_symbols[index].isActive = fibData.isValid;
}

//+------------------------------------------------------------------+
//| Calculate Point of Interest                                      |
//+------------------------------------------------------------------+
void CalculatePointOfInterest(int index)
{
    if(!IsValidIndex(index, "CalculatePOI")) return;
    
    string symbol = g_symbols[index].symbol;
    double keyLevel = g_symbols[index].keyLevel;
    int libIndex = g_symbols[index].libraryIndex;
    
    if(keyLevel <= 0 || libIndex < 0 || libIndex >= MAX_SYMBOLS)
    {
        g_symbols[index].pointOfInterest = 0.0;
        return;
    }
    
    // Get Fibonacci from library
    FibonacciData fibData = g_symbolData[libIndex].fibData;
    
    if(fibData.isValid)
    {
        // Find nearest Fib level to KeyLevel
        double fibLevels[] = {fibData.fib382, fibData.fib500, fibData.fib618};
        double nearestFib = 0.0;
        double minDistance = DBL_MAX;
        
        for(int i = 0; i < 3; i++)
        {
            if(fibLevels[i] <= 0) continue;
            
            double distance = MathAbs(fibLevels[i] - keyLevel);
            if(distance < minDistance)
            {
                minDistance = distance;
                nearestFib = fibLevels[i];
            }
        }
        
        if(nearestFib > 0)
        {
            g_symbols[index].pointOfInterest = (keyLevel + nearestFib) / 2.0;
            return;
        }
    }
    
    // Fallback: use KeyLevel
    g_symbols[index].pointOfInterest = keyLevel;
}

//+------------------------------------------------------------------+
//| Draw Key Level                                                    |
//+------------------------------------------------------------------+
void DrawKeyLevel(int index)
{
    if(!IsValidIndex(index, "DrawKeyLevel")) return;
    
    string symbol = g_symbols[index].symbol;
    double keyLevel = g_symbols[index].keyLevel;
    
    if(keyLevel <= 0) return;
    
    string lineName = "KeyLevel_" + symbol;
    ObjectDelete(0, lineName);
    
    if(ObjectCreate(0, lineName, OBJ_HLINE, 0, 0, keyLevel))
    {
        ObjectSetInteger(0, lineName, OBJPROP_COLOR, clrDodgerBlue);
        ObjectSetInteger(0, lineName, OBJPROP_STYLE, STYLE_DASHDOTDOT);
        ObjectSetInteger(0, lineName, OBJPROP_WIDTH, 2);
        ObjectSetString(0, lineName, OBJPROP_TEXT, symbol + " Key Level");
    }
}

//+------------------------------------------------------------------+
//| Draw Point of Interest                                           |
//+------------------------------------------------------------------+
void DrawPointOfInterest(int index)
{
    if(!IsValidIndex(index, "DrawPOI")) return;
    
    string symbol = g_symbols[index].symbol;
    double poi = g_symbols[index].pointOfInterest;
    
    if(poi <= 0) return;
    
    string lineName = "PointOfInterest_" + symbol;
    ObjectDelete(0, lineName);
    
    if(ObjectCreate(0, lineName, OBJ_HLINE, 0, 0, poi))
    {
        ObjectSetInteger(0, lineName, OBJPROP_COLOR, clrOrange);
        ObjectSetInteger(0, lineName, OBJPROP_STYLE, STYLE_SOLID);
        ObjectSetInteger(0, lineName, OBJPROP_WIDTH, 3);
        ObjectSetString(0, lineName, OBJPROP_TEXT, symbol + " POI");
    }
}

//+------------------------------------------------------------------+
//| Draw Fibonacci                                                    |
//+------------------------------------------------------------------+
void DrawFibonacci(int index)
{
    if(!IsValidIndex(index, "DrawFib")) return;
    
    int libIndex = g_symbols[index].libraryIndex;
    if(libIndex < 0 || libIndex >= MAX_SYMBOLS) return;
    
    string symbol = g_symbols[index].symbol;
    
    // Get Fibonacci data from library
    FibonacciData fibData = g_symbolData[libIndex].fibData;
    
    if(!fibData.isValid) return;
    
    // Draw using library function
    DrawPolishedFibonacci(symbol, fibData, 0);
}

//+------------------------------------------------------------------+
//| Get Trading Signal                                                |
//+------------------------------------------------------------------+
int GetTradingSignal(int index)
{
    if(!IsValidIndex(index, "GetSignal")) return 0;
    
    string symbol = g_symbols[index].symbol;
    double poi = g_symbols[index].pointOfInterest;
    int libIndex = g_symbols[index].libraryIndex;
    
    if(poi <= 0 || libIndex < 0 || libIndex >= MAX_SYMBOLS) return 0;
    
    // Check if price near POI
    if(!IsPriceNearLevel(symbol, poi, KeyLevelTolerance))
        return 0;
    
    // Get Fibonacci trend
    FibonacciData fibData = g_symbolData[libIndex].fibData;
    
    if(!fibData.isValid) return 0;
    
    // Check for rejection if required
    if(RequireRejectionPattern)
    {
        RejectionSignal rejection = PricePosition(symbol, poi, PatternTimeframe);
        
        if(fibData.isBullishTrend && rejection.isBullishRejection)
        {
            PrintFormat("✅ %s: BUY - Bullish Fib + %s", symbol, rejection.rejectionType);
            return 1;
        }
        
        if(fibData.isBearishTrend && rejection.isBearishRejection)
        {
            PrintFormat("✅ %s: SELL - Bearish Fib + %s", symbol, rejection.rejectionType);
            return -1;
        }
        
        return 0;
    }
    
    // No rejection required - use Fib trend only
    if(fibData.isBullishTrend)
    {
        PrintFormat("✅ %s: BUY - Bullish Fibonacci", symbol);
        return 1;
    }
    
    if(fibData.isBearishTrend)
    {
        PrintFormat("✅ %s: SELL - Bearish Fibonacci", symbol);
        return -1;
    }
    
    return 0;
}

//+------------------------------------------------------------------+
//| Check Trade Cooldown                                             |
//+------------------------------------------------------------------+
bool CheckTradeCooldown(int index)
{
    if(!IsValidIndex(index, "CheckCooldown")) return false;
    
    if(g_symbols[index].lastTradeTime == 0) return true;
    
    datetime timeSinceLast = TimeCurrent() - g_symbols[index].lastTradeTime;
    int cooldownSeconds = MinutesBetweenTrades * 60;
    
    return (timeSinceLast >= cooldownSeconds);
}

//+------------------------------------------------------------------+
//| Get Deinitialization Reason                                      |
//+------------------------------------------------------------------+
string GetDeinitReasonText(int reason)
{
    switch(reason)
    {
        case REASON_REMOVE:      return "EA removed";
        case REASON_RECOMPILE:   return "Recompiled";
        case REASON_CHARTCHANGE: return "Chart changed";
        case REASON_CHARTCLOSE:  return "Chart closed";
        case REASON_PARAMETERS:  return "Parameters changed";
        case REASON_ACCOUNT:     return "Account changed";
        case REASON_TEMPLATE:    return "Template applied";
        case REASON_INITFAILED:  return "Init failed";
        case REASON_CLOSE:       return "Terminal closed";
        default:                 return "Unknown";
    }
}
