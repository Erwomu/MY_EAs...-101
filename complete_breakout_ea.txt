void InitializeAdvancedRiskManagement()
{
    if(!UseAdvancedRisk)

        return;
    }
    
    g_advInitialBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    g_advPeakBalance = g_advInitialBalance;
    g_advAdjustedRisk = RiskPercent;
    
    Print("========================================");
    Print("Advanced Risk Management: ENABLED");
    PrintFormat("  Initial Balance: $%.2f", g_advInitialBalance);
    PrintFormat("  Base Risk: %.2f%%", RiskPercent);
    PrintFormat("  Drawdown Defense: %s (trigger: %.1f%%)", 
               UseDrawdownDefense ? "ON" : "OFF", DrawdownThreshold);
    PrintFormat("  Profit Accelerator: %s (trigger: %.1f%%)", 
               UseProfitAccelerator ? "ON" : "OFF", ProfitThreshold);
    Print("========================================");
}

//+------------------------------------------------------------------+
//| Calculate Dynamic Risk Based on Account State                    |
//+------------------------------------------------------------------+
double CalculateDynamicRiskPercentage()
{
    if(!UseAdvancedRisk)
        return RiskPercent;
    
    double currentBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    
    if(currentBalance > g_advPeakBalance)
        g_advPeakBalance = currentBalance;
    
    if(g_advInitialBalance > 0)
        g_advCurrentProfit = ((currentBalance - g_advInitialBalance) / g_advInitialBalance) * 100.0;
    
    if(g_advPeakBalance > 0)
        g_advCurrentDrawdown = ((g_advPeakBalance - currentBalance) / g_advPeakBalance) * 100.0;
    
    double adjustedRisk = RiskPercent;
    
    if(UseDrawdownDefense && g_advCurrentDrawdown >= DrawdownThreshold)
    {
        adjustedRisk *= DrawdownRiskMultiplier;
        
        static datetime lastDrawdownAlert = 0;
        if(TimeCurrent() - lastDrawdownAlert > 3600)
        {
            PrintFormat("🛡️ DRAWDOWN DEFENSE: %.2f%% drawdown | Risk: %.2f%% → %.2f%%", 
                       g_advCurrentDrawdown, RiskPercent, adjustedRisk);
            lastDrawdownAlert = TimeCurrent();
        }
    }
    else if(UseProfitAccelerator && g_advCurrentProfit >= ProfitThreshold)
    {
        adjustedRisk *= ProfitRiskMultiplier;
        
        static datetime lastProfitAlert = 0;
        if(TimeCurrent() - lastProfitAlert > 3600)
        {
            PrintFormat("🚀 PROFIT ACCELERATOR: +%.2f%% profit | Risk: %.2f%% → %.2f%%", 
                       g_advCurrentProfit, RiskPercent, adjustedRisk);
            lastProfitAlert = TimeCurrent();
        }
    }
    
    adjustedRisk = MathMin(adjustedRisk, MaxRiskPercent);
    adjustedRisk = MathMax(adjustedRisk, 0.1);
    
    g_advAdjustedRisk = adjustedRisk;
    return adjustedRisk;
}

//+------------------------------------------------------------------+
//| Get Advanced Risk Status                                         |
//+------------------------------------------------------------------+
string GetAdvancedRiskStatus()
{
    if(!UseAdvancedRisk)
        return "";
        
    string status = "";
    
    if(g_advCurrentDrawdown >= DrawdownThreshold && UseDrawdownDefense)
    {
        status = StringFormat(" | 🛡️ DEFENSE (%.1f%% DD)", g_advCurrentDrawdown);
    }
    else if(g_advCurrentProfit >= ProfitThreshold && UseProfitAccelerator)
    {
        status = StringFormat(" | 🚀 ACCELERATOR (+%.1f%%)", g_advCurrentProfit);
    }
    
    return status;
}

//+------------------------------------------------------------------+
//| Expert initialization function                                   |
//+------------------------------------------------------------------+
int OnInit()
{
    Print("========================================");
    Print("Initializing Breakout EA v2.10");
    Print("========================================");
    
    g_eaTrade.SetExpertMagicNumber(MagicNumber);
    g_eaTrade.SetDeviationInPoints(10);
    g_eaTrade.SetTypeFilling(ORDER_FILLING_IOC);
    g_eaTrade.SetAsyncMode(false);
    g_eaTrade.LogLevel(LOG_LEVEL_ERRORS);
    
    g_symbols[0] = Symbol1;
    g_symbols[1] = Symbol2;
    g_symbols[2] = Symbol3;
    
    int validSymbols = 0;
    for(int i = 0; i < 3; i++)
    {
        g_eaATRHandles[i] = INVALID_HANDLE;
        
        if(g_symbols[i] == "" || g_symbols[i] == "NONE")
        {
            g_symbols[i] = "";
            continue;
        }
        
        if(!SymbolInfoInteger(g_symbols[i], SYMBOL_SELECT))
        {
            if(!SymbolSelect(g_symbols[i], true))
            {
                PrintFormat("ERROR: Symbol %s not found", g_symbols[i]);
                g_symbols[i] = "";
                continue;
            }
        }
        
        g_eaATRHandles[i] = iATR(g_symbols[i], LTF_Period, ATRPeriod);
        if(g_eaATRHandles[i] == INVALID_HANDLE)
        {
            PrintFormat("ERROR: Failed to create ATR handle for %s", g_symbols[i]);
            g_symbols[i] = "";
            continue;
        }
        
        Sleep(100);
        
        double atrTest[];
        if(CopyBuffer(g_eaATRHandles[i], 0, 0, 1, atrTest) <= 0)
        {
            PrintFormat("ERROR: ATR indicator not ready for %s", g_symbols[i]);
            IndicatorRelease(g_eaATRHandles[i]);
            g_eaATRHandles[i] = INVALID_HANDLE;
            g_symbols[i] = "";
            continue;
        }
        
        g_htfData[i].Reset();
        g_stats[i].Reset();
        
        if(!MarkPreviousHTF(g_symbols[i], i))
        {
            PrintFormat("WARNING: Failed to mark HTF levels for %s", g_symbols[i]);
        }
        
        validSymbols++;
        PrintFormat("Symbol initialized: %s (ATR Handle: %d)", g_symbols[i], g_eaATRHandles[i]);
    }
    
    if(validSymbols == 0)
    {
        Print("ERROR: No valid symbols to trade");
        return INIT_FAILED;
    }
    
    PrintFormat("%d symbol(s) ready for trading", validSymbols);
    
    // Initialize advanced risk management
    InitializeAdvancedRiskManagement();
    
    if(ShowDashboard)
    {
        if(!CreateDashboard())
        {
            Print("WARNING: Failed to create dashboard");
        }
    }
    
    EventSetTimer(60);
    
    g_initialized = true;
    g_lastCheck = TimeCurrent();
    
    Print("========================================");
    Print("Breakout EA initialized successfully!");
    Print("========================================");
    
    return INIT_SUCCEEDED;
}

//+------------------------------------------------------------------+
//| Expert deinitialization function                                 |
//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
    Print("========================================");
    Print("Deinitializing Breakout EA");
    PrintFormat("Reason: %s", GetDeinitReasonText(reason));
    Print("========================================");
    
    EventKillTimer();
    
    for(int i = 0; i < 3; i++)
    {
        if(g_eaATRHandles[i] != INVALID_HANDLE)
        {
            IndicatorRelease(g_eaATRHandles[i]);
            g_eaATRHandles[i] = INVALID_HANDLE;
        }
    }
    
    if(ShowDashboard)
    {
        RemoveDashboard();
    }
    
    for(int i = 0; i < 3; i++)
    {
        if(g_symbols[i] != "")
        {
            ObjectDelete(0, "HTF_High_" + g_symbols[i]);
            ObjectDelete(0, "HTF_Low_" + g_symbols[i]);
            ObjectDelete(0, "HTF_Range_" + g_symbols[i]);
        }
    }
    
    ChartRedraw();
    Print("Breakout EA deinitialized successfully");
}

//+------------------------------------------------------------------+
//| Calculate Lot Size with Enhanced Risk Management                |
//+------------------------------------------------------------------+
double CalculateLotSize(const string symbol, const int index, const double atr)
{
    if(!g_eaSymbolInfo.Name(symbol) || atr <= 0)
        return g_eaSymbolInfo.LotsMin();
    
    // Get dynamic risk percentage
    double riskPercent = CalculateDynamicRiskPercentage();
    
    double accountBalance = AccountInfoDouble(ACCOUNT_BALANCE);
    double riskAmount = (riskPercent / 100.0) * accountBalance;
    
    double stopLossDistance = atr * ATRStopLoss;
    double tickValue = g_eaSymbolInfo.TickValue();
    double tickSize = g_eaSymbolInfo.TickSize();
    
    if(tickValue <= 0 || tickSize <= 0 || stopLossDistance <= 0)
        return g_eaSymbolInfo.LotsMin();
    
    double stopLossInTicks = stopLossDistance / tickSize;
    double lotSize = riskAmount / (stopLossInTicks * tickValue);
    
    double minLot = g_eaSymbolInfo.LotsMin();
    double maxLot = g_eaSymbolInfo.LotsMax();
    double lotStep = g_eaSymbolInfo.LotsStep();
    
    lotSize = MathMax(minLot, MathMin(lotSize, maxLot));
    lotSize = NormalizeDouble(MathFloor(lotSize / lotStep) * lotStep, 2);
    
    if(UseAdvancedRisk && riskPercent != RiskPercent)
    {
        PrintFormat("%s: Risk %.2f%% (Base: %.2f%%) | Lot: %.2f | $%.2f at risk",
                   symbol, riskPercent, RiskPercent, lotSize, riskAmount);
    }
    
    return lotSize;
}

//+------------------------------------------------------------------+
//| Get ATR Value                                                    |
//+------------------------------------------------------------------+
double GetATRValue(const int index)
{
    if(g_eaATRHandles[index] == INVALID_HANDLE)
        return 0;
    
    double atrBuffer[];
    ArraySetAsSeries(atrBuffer, true);
    
    if(CopyBuffer(g_eaATRHandles[index], 0, 0, 1, atrBuffer) <= 0)
    {
        PrintFormat("ERROR: Failed to copy ATR buffer for index %d", index);
        return 0;
    }
    
    return atrBuffer[0];
}

//+------------------------------------------------------------------+
//| Manage Positions                                                 |
//+------------------------------------------------------------------+
void ManagePositions(const string symbol, const int index)
{
    double atr = GetATRValue(index);
    if(atr <= 0)
        return;
    
    for(int i = PositionsTotal() - 1; i >= 0; i--)
    {
        if(!g_eaPosition.SelectByIndex(i))
            continue;
        
        if(g_eaPosition.Symbol() != symbol || g_eaPosition.Magic() != MagicNumber)
            continue;
        
        if(!g_eaSymbolInfo.Name(symbol))
            continue;
        
        g_eaSymbolInfo.RefreshRates();
        
        double currentSL = g_eaPosition.StopLoss();
        double openPrice = g_eaPosition.PriceOpen();
        int digits = (int)g_eaSymbolInfo.Digits();
        
        bool modified = false;
        double newSL = 0.0;
        
        if(g_eaPosition.PositionType() == POSITION_TYPE_BUY)
        {
            double bid = g_eaSymbolInfo.Bid();
            double profitDistance = bid - openPrice;
            
            if(profitDistance >= (atr * BEMultiplier) && currentSL < openPrice)
            {
                newSL = NormalizeDouble(openPrice + (g_eaSymbolInfo.Point() * 10), digits);
                modified = true;
                PrintFormat("%s BUY: Moving to Break-Even at %.5f", symbol, newSL);
            }
            else if(profitDistance >= (atr * TrailStartMultiplier) && currentSL >= openPrice)
            {
                double trailSL = NormalizeDouble(bid - (atr * TrailStopMultiplier), digits);
                if(trailSL > currentSL)
                {
                    newSL = trailSL;
                    modified = true;
                    PrintFormat("%s BUY: Trailing SL to %.5f", symbol, newSL);
                }
            }
        }
        else if(g_eaPosition.PositionType() == POSITION_TYPE_SELL)
        {
            double ask = g_eaSymbolInfo.Ask();
            double profitDistance = openPrice - ask;
            
            if(profitDistance >= (atr * BEMultiplier) && (currentSL == 0 || currentSL > openPrice))
            {
                newSL = NormalizeDouble(openPrice - (g_eaSymbolInfo.Point() * 10), digits);
                modified = true;
                PrintFormat("%s SELL: Moving to Break-Even at %.5f", symbol, newSL);
            }
            else if(profitDistance >= (atr * TrailStartMultiplier) && currentSL > 0 && currentSL <= openPrice)
            {
                double trailSL = NormalizeDouble(ask + (atr * TrailStopMultiplier), digits);
                if(currentSL == 0 || trailSL < currentSL)
                {
                    newSL = trailSL;
                    modified = true;
                    PrintFormat("%s SELL: Trailing SL to %.5f", symbol, newSL);
                }
            }
        }
        
        if(modified && newSL > 0)
        {
            if(g_eaTrade.PositionModify(g_eaPosition.Ticket(), newSL, g_eaPosition.TakeProfit()))
            {
                if(g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
                {
                    PrintFormat("%s: SL modified successfully to %.5f", symbol, newSL);
                }
            }
            else
            {
                PrintFormat("%s: Failed to modify SL | Error:%d", symbol, g_eaTrade.ResultRetcode());
            }
        }
    }
}

//+------------------------------------------------------------------+
//| Check Trading Session                                            |
//+------------------------------------------------------------------+
bool IsInTradingSession()
{
    MqlDateTime current;
    TimeToStruct(TimeCurrent(), current);
    int currentHour = current.hour;
    
    if(currentHour >= LondonStart && currentHour < LondonEnd)
        return true;
    
    if(currentHour >= NewYorkStart && currentHour < NewYorkEnd)
        return true;
    
    return false;
}

//+------------------------------------------------------------------+
//| Check Weekend Time                                               |
//+------------------------------------------------------------------+
bool IsWeekendTime()
{
    MqlDateTime current;
    TimeToStruct(TimeCurrent(), current);
    
    if(current.day_of_week == 5 && current.hour >= 20)
        return true;
    
    if(current.day_of_week == 0 && current.hour < 21)
        return true;
    
    return false;
}

//+------------------------------------------------------------------+
//| Check Spread                                                     |
//+------------------------------------------------------------------+
bool IsSpreadAcceptable(const string symbol)
{
    if(!g_eaSymbolInfo.Name(symbol))
        return false;
    
    g_eaSymbolInfo.RefreshRates();
    
    double spread = g_eaSymbolInfo.Spread() * g_eaSymbolInfo.Point();
    double maxSpread = MaxSpreadPips * g_eaSymbolInfo.Point() * 10;
    
    if(spread > maxSpread)
    {
        PrintFormat("%s: Spread %.1f pips exceeds maximum %.1f pips", 
                   symbol, spread / (g_eaSymbolInfo.Point() * 10), MaxSpreadPips);
        return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Update Statistics                                                |
//+------------------------------------------------------------------+
void UpdateStatistics()
{
    if(!HistorySelect(0, TimeCurrent()))
        return;
    
    for(int i = 0; i < 3; i++)
    {
        if(g_symbols[i] == "")
            continue;
        
        double totalProfit = 0;
        int totalTrades = 0;
        int wins = 0;
        int losses = 0;
        int consecLosses = 0;
        
        for(int j = HistoryDealsTotal() - 1; j >= 0; j--)
        {
            ulong ticket = HistoryDealGetTicket(j);
            if(ticket == 0)
                continue;
            
            if(HistoryDealGetString(ticket, DEAL_SYMBOL) != g_symbols[i])
                continue;
            
            if(HistoryDealGetInteger(ticket, DEAL_MAGIC) != MagicNumber)
                continue;
            
            if(HistoryDealGetInteger(ticket, DEAL_ENTRY) != DEAL_ENTRY_OUT)
                continue;
            
            double profit = HistoryDealGetDouble(ticket, DEAL_PROFIT);
            totalProfit += profit;
            totalTrades++;
            
            if(profit > 0)
            {
                wins++;
                consecLosses = 0;
            }
            else if(profit < 0)
            {
                losses++;
                consecLosses++;
            }
            
            if(consecLosses > g_stats[i].consecutiveLosses)
                g_stats[i].consecutiveLosses = consecLosses;
        }
        
        g_stats[i].totalProfit = totalProfit;
        g_stats[i].totalTrades = totalTrades;
        g_stats[i].winningTrades = wins;
        g_stats[i].losingTrades = losses;
    }
}

//+------------------------------------------------------------------+
//| Update Dashboard                                                 |
//+------------------------------------------------------------------+
void UpdateDashboard()
{
    double balance = AccountInfoDouble(ACCOUNT_BALANCE);
    double equity = AccountInfoDouble(ACCOUNT_EQUITY);
    double profit = equity - balance;
    int totalPos = GetAllPositionsCount();
    
    string riskStatus = GetAdvancedRiskStatus();
    
    string accountText = StringFormat("Balance: $%.2f | Equity: $%.2f | P/L: $%.2f | Pos: %d/%d%s",
                                     balance, equity, profit, totalPos, MaxTotalPositions, riskStatus);
    ObjectSetString(0, "Dashboard_Account", OBJPROP_TEXT, accountText);
    ObjectSetInteger(0, "Dashboard_Account", OBJPROP_COLOR, profit >= 0 ? clrLime : clrRed);
    
    bool inSession = IsInTradingSession();
    string sessionText = StringFormat("Session: %s | HTF: %s | LTF: %s",
                                     inSession ? "ACTIVE" : "CLOSED",
                                     EnumToString(HTF_Period),
                                     EnumToString(LTF_Period));
    ObjectSetString(0, "Dashboard_Session", OBJPROP_TEXT, sessionText);
    ObjectSetInteger(0, "Dashboard_Session", OBJPROP_COLOR, inSession ? clrLime : clrOrange);
    
    // Update risk state
    if(UseAdvancedRisk)
    {
        string riskStateText = "";
        color riskColor = clrWhite;
        
        if(g_advCurrentDrawdown >= DrawdownThreshold && UseDrawdownDefense)
        {
            riskStateText = StringFormat("Risk: %.2f%% | 🛡️ DEFENSE MODE (%.1f%% DD)", 
                                       g_advAdjustedRisk, g_advCurrentDrawdown);
            riskColor = clrOrange;
        }
        else if(g_advCurrentProfit >= ProfitThreshold && UseProfitAccelerator)
        {
            riskStateText = StringFormat("Risk: %.2f%% | 🚀 ACCELERATOR (+%.1f%%)", 
                                       g_advAdjustedRisk, g_advCurrentProfit);
            riskColor = clrLime;
        }
        else
        {
            riskStateText = StringFormat("Risk: %.2f%% | ⚡ NORMAL MODE", g_advAdjustedRisk);
            riskColor = clrWhite;
        }
        
        ObjectSetString(0, "Dashboard_RiskState", OBJPROP_TEXT, riskStateText);
        ObjectSetInteger(0, "Dashboard_RiskState", OBJPROP_COLOR, riskColor);
    }
    
    for(int i = 0; i < 3; i++)
    {
        string labelName = "Dashboard_Symbol_" + IntegerToString(i);
        
        if(g_symbols[i] == "")
        {
            ObjectSetString(0, labelName, OBJPROP_TEXT, "");
            continue;
        }
        
        string status = "-";
        color textColor = clrGray;
        
        int positions = GetSymbolPositionsCount(g_symbols[i]);
        
        if(positions > 0)
        {
            double symbolProfit = 0;
            for(int j = 0; j < PositionsTotal(); j++)
            {
                if(g_eaPosition.SelectByIndex(j))
                {
                    if(g_eaPosition.Symbol() == g_symbols[i] && g_eaPosition.Magic() == MagicNumber)
                    {
                        symbolProfit += g_eaPosition.Profit() + g_eaPosition.Swap() + g_eaPosition.Commission();
                    }
                }
            }
            
            status = StringFormat("%.2f", symbolProfit);
            textColor = symbolProfit >= 0 ? clrLime : clrRed;
        }
        else if(g_htfData[i].breakoutUp)
        {
            status = "UP";
            textColor = clrAqua;
        }
        else if(g_htfData[i].breakoutDown)
        {
            status = "DOWN";
            textColor = clrOrange;
        }
        else if(!g_stats[i].tradingEnabled)
        {
            status = "PAUSED";
            textColor = clrRed;
        }
        else
        {
            status = "WAITING";
            textColor = clrGray;
        }
        
        string winRate = g_stats[i].totalTrades > 0 ? 
                        StringFormat("%.0f%%", (double)g_stats[i].winningTrades / g_stats[i].totalTrades * 100) : 
                        "-";
        
        string text = StringFormat("%s: %s | W/L:%d/%d(%s)", 
                                  g_symbols[i], status,
                                  g_stats[i].winningTrades, 
                                  g_stats[i].losingTrades,
                                  winRate);
        
        ObjectSetString(0, labelName, OBJPROP_TEXT, text);
        ObjectSetInteger(0, labelName, OBJPROP_COLOR, textColor);
    }
    
    ChartRedraw();
}

{

        case REASON_REMOVE: return "EA removed from chart";
        case REASON_RECOMPILE: return "EA recompiled";
        case REASON_CHARTCHANGE: return "Chart symbol/period changed";
        case REASON_CHARTCLOSE: return "Chart closed";
        case REASON_PARAMETERS: return "Input parameters changed";
        case REASON_ACCOUNT: return "Account changed";
        case REASON_TEMPLATE: return "Template applied";
        case REASON_INITFAILED: return "Initialization failed";
        case REASON_CLOSE: return "Terminal closing";
        default: return "Unknown reason";
    }
}

//+------------------------------------------------------------------+
//| Expert tick function                                             |
//+------------------------------------------------------------------+
void OnTick()
{
    if(!g_initialized)
        return;
    
    datetime currentTime = TimeCurrent();
    if(currentTime - g_lastCheck < 60)
        return;
    
    g_lastCheck = currentTime;
    
    for(int i = 0; i < 3; i++)
    {
        if(g_symbols[i] == "")
            continue;
        
        ProcessSymbol(g_symbols[i], i);
    }
    
    if(ShowDashboard)
    {
        UpdateDashboard();
    }
}

//+------------------------------------------------------------------+
//| Timer function                                                   |
//+------------------------------------------------------------------+
void OnTimer()
{
    if(!g_initialized)
        return;
    
    for(int i = 0; i < 3; i++)
    {
        if(g_symbols[i] != "")
        {
            CheckNewHTFCandle(g_symbols[i], i);
        }
    }
    
    UpdateStatistics();
    
    if(ShowDashboard)
    {
        UpdateDashboard();
    }
}

//+------------------------------------------------------------------+
//| Process Symbol Trading Logic                                     |
//+------------------------------------------------------------------+
void ProcessSymbol(const string symbol, const int index)
{
    if(!g_stats[index].tradingEnabled)
    {
        if(TimeCurrent() >= g_stats[index].pausedUntil)
        {
            g_stats[index].tradingEnabled = true;
            g_stats[index].consecutiveLosses = 0;
            PrintFormat("%s: Trading resumed", symbol);
        }
        else
        {
            return;
        }
    }
    
    if(UseSessionFilter && !IsInTradingSession())
        return;
    
    if(AvoidWeekends && IsWeekendTime())
        return;
    
    if(g_stats[index].consecutiveLosses >= MaxConsecutiveLosses)
    {
        g_stats[index].tradingEnabled = false;
        g_stats[index].pausedUntil = iTime(symbol, PERIOD_D1, 0) + 86400;
        PrintFormat("%s: Trading paused due to %d consecutive losses", 
                   symbol, g_stats[index].consecutiveLosses);
        return;
    }
    
    int currentPositions = GetSymbolPositionsCount(symbol);
    if(currentPositions >= MaxOpenPositions)
    {
        ManagePositions(symbol, index);
        return;
    }
    
    int totalPositions = GetAllPositionsCount();
    if(totalPositions >= MaxTotalPositions)
    {
        ManagePositions(symbol, index);
        return;
    }
    
    if(currentPositions > 0)
    {
        ManagePositions(symbol, index);
        return;
    }
    
    CheckNewHTFCandle(symbol, index);
    
    if(!g_htfData[index].tradeExecuted && CheckBreakout(symbol, index))
    {
        if(CheckEntryConfirmation(symbol, index))
        {
            if(UseSpreadFilter && !IsSpreadAcceptable(symbol))
            {
                PrintFormat("%s: Spread too high, skipping trade", symbol);
                return;
            }
            
            if(ExecuteBreakoutTrade(symbol, index))
            {
                g_htfData[index].tradeExecuted = true;
                g_stats[index].lastTradeTime = TimeCurrent();
            }
        }
    }
}

//+------------------------------------------------------------------+
//| Check for New HTF Candle                                         |
//+------------------------------------------------------------------+
void CheckNewHTFCandle(const string symbol, const int index)
{
    datetime currentHTF = iTime(symbol, HTF_Period, 0);
    
    if(currentHTF != g_htfData[index].time)
    {
        if(g_htfData[index].time > 0)
        {
            PrintFormat("%s: New %s candle detected", symbol, EnumToString(HTF_Period));
        }
        
        MarkPreviousHTF(symbol, index);
    }
}

//+------------------------------------------------------------------+
//| Mark Previous HTF Levels                                         |
//+------------------------------------------------------------------+
bool MarkPreviousHTF(const string symbol, const int index)
{
    double open[], close[], high[], low[];
    datetime time[];
    
    if(CopyOpen(symbol, HTF_Period, 1, 1, open) <= 0 ||
       CopyClose(symbol, HTF_Period, 1, 1, close) <= 0 ||
       CopyHigh(symbol, HTF_Period, 1, 1, high) <= 0 ||
       CopyLow(symbol, HTF_Period, 1, 1, low) <= 0 ||
       CopyTime(symbol, HTF_Period, 0, 1, time) <= 0)
    {
        PrintFormat("ERROR: Failed to copy HTF data for %s", symbol);
        return false;
    }
    
    g_htfData[index].open = open[0];
    g_htfData[index].close = close[0];
    g_htfData[index].high = high[0];
    g_htfData[index].low = low[0];
    g_htfData[index].time = time[0];
    g_htfData[index].breakoutUp = false;
    g_htfData[index].breakoutDown = false;
    g_htfData[index].confirmationCandles = 0;
    g_htfData[index].tradeExecuted = false;
    
    if(symbol == _Symbol)
    {
        DrawHTFLevels(symbol, index);
    }
    
    PrintFormat("%s HTF: H:%.5f L:%.5f O:%.5f C:%.5f", 
               symbol, g_htfData[index].high, g_htfData[index].low, 
               g_htfData[index].open, g_htfData[index].close);
    
    return true;
}

//+------------------------------------------------------------------+
//| Check for Breakout                                               |
//+------------------------------------------------------------------+
bool CheckBreakout(const string symbol, const int index)
{
    double close[];
    if(CopyClose(symbol, LTF_Period, 0, 1, close) <= 0)
        return false;
    
    double currentPrice = close[0];
    
    if(currentPrice > g_htfData[index].high && !g_htfData[index].breakoutUp)
    {
        g_htfData[index].breakoutUp = true;
        g_htfData[index].breakLevel = g_htfData[index].high;
        g_htfData[index].confirmationCandles = 0;
        PrintFormat("%s: BREAKOUT ABOVE HTF High %.5f", symbol, g_htfData[index].high);
        return true;
    }
    
    if(currentPrice < g_htfData[index].low && !g_htfData[index].breakoutDown)
    {
        g_htfData[index].breakoutDown = true;
        g_htfData[index].breakLevel = g_htfData[index].low;
        g_htfData[index].confirmationCandles = 0;
        PrintFormat("%s: BREAKOUT BELOW HTF Low %.5f", symbol, g_htfData[index].low);
        return true;
    }
    
    return g_htfData[index].breakoutUp || g_htfData[index].breakoutDown;
}

//+------------------------------------------------------------------+
//| Check Entry Confirmation                                         |
//+------------------------------------------------------------------+
bool CheckEntryConfirmation(const string symbol, const int index)
{
    if(UseTwoCandles)
    {
        if(CheckTwoCandlesConfirmation(symbol, index))
        {
            PrintFormat("%s: Two candles confirmation passed", symbol);
            return true;
        }
    }
    
    if(UseCandlePatterns)
    {
        if(CheckCandlePatternConfirmation(symbol, index))
        {
            PrintFormat("%s: Candle pattern confirmation passed", symbol);
            return true;
        }
    }
    
    if(!UseTwoCandles && !UseCandlePatterns)
    {
        return true;
    }
    
    return false;
}

//+------------------------------------------------------------------+
//| Check Two Candles Confirmation                                   |
//+------------------------------------------------------------------+
bool CheckTwoCandlesConfirmation(const string symbol, const int index)
{
    double close[];
    if(CopyClose(symbol, LTF_Period, 1, 2, close) < 2)
        return false;
    
    double close1 = close[1];
    double close2 = close[0];
    
    if(g_htfData[index].breakoutUp)
    {
        return (close1 > g_htfData[index].breakLevel && close2 > g_htfData[index].breakLevel);
    }
    else if(g_htfData[index].breakoutDown)
    {
        return (close1 < g_htfData[index].breakLevel && close2 < g_htfData[index].breakLevel);
    }
    
    return false;
}

//+------------------------------------------------------------------+
//| Check Candle Pattern Confirmation                                |
//+------------------------------------------------------------------+
bool CheckCandlePatternConfirmation(const string symbol, const int index)
{
    double open[], close[], high[], low[];
    
    if(CopyOpen(symbol, LTF_Period, 1, 2, open) < 2 ||
       CopyClose(symbol, LTF_Period, 1, 2, close) < 2 ||
       CopyHigh(symbol, LTF_Period, 1, 2, high) < 2 ||
       CopyLow(symbol, LTF_Period, 1, 2, low) < 2)
    {
        return false;
    }
    
    double open1 = open[1], close1 = close[1], high1 = high[1], low1 = low[1];
    double open2 = open[0], close2 = close[0], high2 = high[0], low2 = low[0];
    
    double body1 = MathAbs(close1 - open1);
    double body2 = MathAbs(close2 - open2);
    double range1 = high1 - low1;
    double range2 = high2 - low2;
    
    if(range1 == 0 || range2 == 0)
        return false;
    
    if(g_htfData[index].breakoutUp)
    {
        if(close1 > open1 && close2 < open2 && 
           close1 > open2 && open1 < close2 &&
           body1 > body2 * 1.2 &&
           body1/range1 > MinCandleBodyRatio)
        {
            return true;
        }
        
        double lowerWick1 = MathMin(open1, close1) - low1;
        if(close1 > open1 && lowerWick1 > body1 * 2 && body1/range1 > MinCandleBodyRatio)
        {
            return true;
        }
    }
    else if(g_htfData[index].breakoutDown)
    {
        if(close1 < open1 && close2 > open2 && 
           close1 < open2 && open1 > close2 &&
           body1 > body2 * 1.2 &&
           body1/range1 > MinCandleBodyRatio)
        {
            return true;
        }
        
        double upperWick1 = high1 - MathMax(open1, close1);
        if(close1 < open1 && upperWick1 > body1 * 2 && body1/range1 > MinCandleBodyRatio)
        {
            return true;
        }
    }
    
    return false;
}

//+------------------------------------------------------------------+
//| Execute Breakout Trade                                           |
//+------------------------------------------------------------------+
bool ExecuteBreakoutTrade(const string symbol, const int index)
{
    if(!g_eaSymbolInfo.Name(symbol))
    {
        PrintFormat("ERROR: Cannot select symbol %s", symbol);
        return false;
    }
    
    g_eaSymbolInfo.RefreshRates();
    
    double atr = GetATRValue(index);
    if(atr <= 0)
    {
        PrintFormat("ERROR: Invalid ATR value for %s", symbol);
        return false;
    }
    
    double lotSize = CalculateLotSize(symbol, index, atr);
    if(lotSize < g_eaSymbolInfo.LotsMin())
    {
        PrintFormat("ERROR: Lot size %.2f below minimum", lotSize);
        return false;
    }
    
    double price, sl, tp;
    ENUM_ORDER_TYPE orderType;
    string direction;
    
    if(g_htfData[index].breakoutUp)
    {
        orderType = ORDER_TYPE_BUY;
        direction = "BUY";
        price = g_eaSymbolInfo.Ask();
        sl = NormalizeDouble(price - (atr * ATRStopLoss), g_eaSymbolInfo.Digits());
        tp = NormalizeDouble(price + (atr * ATRTakeProfit), g_eaSymbolInfo.Digits());
    }
    else if(g_htfData[index].breakoutDown)
    {
        orderType = ORDER_TYPE_SELL;
        direction = "SELL";
        price = g_eaSymbolInfo.Bid();
        sl = NormalizeDouble(price + (atr * ATRStopLoss), g_eaSymbolInfo.Digits());
        tp = NormalizeDouble(price - (atr * ATRTakeProfit), g_eaSymbolInfo.Digits());
    }
    else
    {
        return false;
    }
    
    double riskDistance = MathAbs(price - sl);
    double rewardDistance = MathAbs(tp - price);
    double rrRatio = rewardDistance / riskDistance;
    
    if(rrRatio < MinRiskReward)
    {
        PrintFormat("%s: R:R ratio %.2f below minimum", symbol, rrRatio);
        return false;
    }
    
    bool result = g_eaTrade.PositionOpen(symbol, orderType, lotSize, price, sl, tp, 
                                        StringFormat("Breakout_%s", direction));
    
    if(result && g_eaTrade.ResultRetcode() == TRADE_RETCODE_DONE)
    {
        PrintFormat("%s %s OPENED | Lot:%.2f | Price:%.5f | SL:%.5f | TP:%.5f | RR:%.2f",
                   symbol, direction, lotSize, price, sl, tp, rrRatio);
        
        g_stats[index].totalTrades++;
        return true;
    }
    else
    {
        PrintFormat("%s: Trade failed | Error:%d", symbol, g_eaTrade.ResultRetcode());
        return false;
    }
}
